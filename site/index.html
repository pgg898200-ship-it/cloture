import cloture

# Stop parallel LLM/agent rollouts the instant consensus is certified (cancels in-flight tasks)
result = await cloture.consensus(
    spawn_fn=lambda i: call_agent_async(prompt, seed=i),
    delta=0.05,
    max_calls=32,
    concurrency=8,
)

# Audit any CSV or Promptfoo eval export directly (or run: cloture audit --csv judges.csv)
